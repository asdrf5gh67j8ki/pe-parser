cmake_minimum_required(VERSION 3.24)
project(pe-analyzer VERSION 1.1.0 LANGUAGES C)
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Windows scanner and live fixture require an x64 build. Use a new build directory with the Visual Studio generator and -A x64, or configure Ninja from an x64 developer shell.")
endif()
include(CTest)
option(PS_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(PS_WARNINGS_AS_ERRORS "Treat project warnings as errors" ON)
add_library(ps_core STATIC src/util.c src/pe.c src/report.c)
target_include_directories(ps_core PUBLIC src)
target_compile_features(ps_core PUBLIC c_std_17)
add_executable(pe-analyzer src/main.c src/scan.c)
target_link_libraries(pe-analyzer PRIVATE ps_core)
if(WIN32)
    target_compile_definitions(pe-analyzer PRIVATE _WIN32_WINNT=0x0a00 PSAPI_VERSION=1)
    target_link_libraries(pe-analyzer PRIVATE psapi)
    add_executable(pe-analyzer-gui WIN32 src/gui.c src/scan.c)
    target_link_libraries(pe-analyzer-gui PRIVATE ps_core psapi user32 gdi32 comctl32 comdlg32 shell32 shlwapi uxtheme dwmapi)
    target_compile_definitions(pe-analyzer-gui PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0a00 PSAPI_VERSION=1)
    if(MINGW)
        target_link_options(pe-analyzer-gui PRIVATE -municode)
    endif()
    if(MINGW)
        target_link_options(pe-analyzer PRIVATE -municode)
    endif()
else()
    target_link_libraries(ps_core PUBLIC m)
endif()
set(ps_targets ps_core pe-analyzer)
if(WIN32)
    list(APPEND ps_targets pe-analyzer-gui)
    install(TARGETS pe-analyzer-gui RUNTIME DESTINATION bin)
endif()
if(BUILD_TESTING)
    add_executable(core_tests tests/core.c)
    target_link_libraries(core_tests PRIVATE ps_core)
    add_test(NAME core COMMAND core_tests)
    list(APPEND ps_targets core_tests)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py" "$<TARGET_FILE:pe-analyzer>")
    add_test(NAME demo_protocol COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/demo_protocol.py")
    if(WIN32)
        add_test(NAME gui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/gui.py" "$<TARGET_FILE:pe-analyzer-gui>")
        add_library(demo_module SHARED tests/demo_module.c)
        add_executable(demo_host tests/demo_host.c)
        if(MINGW)
            target_link_options(demo_host PRIVATE -municode)
        endif()
        list(APPEND ps_targets demo_module demo_host)
        add_test(NAME live COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/live.py" "$<TARGET_FILE:pe-analyzer>" "$<TARGET_FILE:demo_host>" "$<TARGET_FILE:demo_module>")
    endif()
endif()
foreach(target IN LISTS ps_targets)
    set_target_properties(${target} PROPERTIES C_EXTENSIONS OFF)
    if(MSVC)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
        if(PS_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(PS_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
    if(PS_SANITIZERS AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
install(TARGETS pe-analyzer RUNTIME DESTINATION bin)
